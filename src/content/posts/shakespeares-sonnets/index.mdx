import { lettersInText, Scrawl } from "./Scrawl";
import { WeeklyScrawl } from "./WeeklyScrawl";
import quickLeft from "./quick-left.scrawl?raw";
import quickRight from "./quick-right.scrawl?raw";

export const texts = Object.values(
  import.meta.glob("./text/*.txt", {
    query: "?raw",
    import: "default",
    eager: true,
  }),
);

export const days = Object.values(
  // only the derived wpm crosses to the client; the stroke data stays here
  import.meta.glob("./scrawl/*.scrawl", {
    query: "?raw",
    import: "default",
    eager: true,
  }),
).map((scrawl, i) => {
  let totalMs = 0;
  for (const line of scrawl.split("\n")) {
    if (!line.trim()) continue;
    for (const point of line.split(";")) {
      const dt = Number(point.split(",")[2]);
      if (!Number.isNaN(dt)) totalMs += dt;
    }
  }
  const wpm = lettersInText(texts[i]) / 5 / (totalMs / 60_000);
  return { wpm };
});

export const scrawlUrls = Object.values(
  import.meta.glob("./scrawl/*.scrawl", {
    query: "?url",
    import: "default",
    eager: true,
  }),
);

export const chosen = [
  2, 12, 18, 26, 29, 42, 44, 52, 62, 65, 71, 79, 91, 94, 99, 106, 116, 124, 130,
  135,
];

With a few minutes of practice a day, you can pick up any number of odd skills. This time, I'm learning to write with my spare hand.

Unlike those poor left-handed children in history books, I, in the present-day, overgrown and right-handed, will willingly master my lazy left. And what a coincidence! Shakespeare has a training arc's worth of sonnets.

Each of these 14 line love poems take around 10 minutes to copy, which is short enough that I have a chance of perservering each day for the next 5 months. I'm using a drawing tablet and my vibe-coded [writing reanimation tool](https://rescrawl.kylezhe.ng) to record my writing and track my improvement in speed (words per minute) and tidiness (subjective).

## Misc

For the words per minute calculation, words are 5 letters and pauses longer than 2 seconds are excluded.

I referenced the Poetry Foundation website to write the first three sonnets, which has different punctuation. For consistency however, all text, and all other writing shown uses The Folger Shakespeare version.
